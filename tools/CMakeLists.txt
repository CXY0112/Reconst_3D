cmake_minimum_required(VERSION 3.16)
project(reconst_bridge LANGUAGES CXX)
find_package(RTABMap REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_library(online_bridge SHARED online_bridge.cpp)
target_compile_features(online_bridge PRIVATE cxx_std_17)
target_include_directories(online_bridge PRIVATE ${RTABMap_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(online_bridge PRIVATE ${RTABMap_LIBRARIES} ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(online_bridge PRIVATE /utf-8 /EHsc)
  target_compile_definitions(online_bridge PRIVATE NOMINMAX _USE_MATH_DEFINES)
endif()
