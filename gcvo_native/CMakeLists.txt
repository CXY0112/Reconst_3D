cmake_minimum_required(VERSION 3.18)
project(reconst_gcvo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RECONST_GCVO_CUDA "Build additional wrapper for the unchanged official CUDA implementation" OFF)
set(GCVO_UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/gcvo/source" CACHE PATH "Official G-CVO checkout")
find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3 REQUIRED)
add_library(gcvo_cpu SHARED gcvo_cpu.cpp)
install(TARGETS gcvo_cpu RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
target_include_directories(gcvo_cpu PRIVATE ${EIGEN3_INCLUDE_DIR})
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(gcvo_cpu PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(gcvo_cpu PRIVATE /O2 /EHsc /utf-8)
else()
  target_compile_options(gcvo_cpu PRIVATE -O3)
endif()
if(RECONST_GCVO_CUDA)
  enable_language(CUDA)
  set(GCVO_BUILD_APPS OFF CACHE BOOL "" FORCE)
  set(GCVO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${GCVO_UPSTREAM} official_gcvo)
  add_library(gcvo_cuda SHARED gcvo_cuda.cpp)
  target_link_libraries(gcvo_cuda PRIVATE gcvo_s3)
  install(TARGETS gcvo_cuda RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
endif()
